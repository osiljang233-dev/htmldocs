```php
       달력이 이걸 그린다. 모양이 다르면 뷰가 두 벌이 되니 반드시 같게 둔다. */
    private function picks_day_counts($days_map) {
        ksort($days_map);
        $out = [];
        foreach ($days_map as $d => $n) {
            $out[] = (object) ['d' => (string) $d, 'cnt' => (int) $n];
        }
        return $out;
    }
```
