## 수정 내용

`$soccerResultsMap` 루프(698행) 바로 위에 아래 블록을 통째로 추가해줘.

```php
/* ── 빅5 리그 순위표 갱신 ───────────────────────────────────────────
   {리그}_standings_YYYY-MM-DD.json → assets/data/{리그}.json 의 standings 배열 교체
   results · fixtures 는 건드리지 않는다. standings 가 10팀 미만이면 반영하지 않는다. */
$soccerStandingsMap = [
    'epl'        => 'epl.json',
    'laliga'     => 'laliga.json',
    'bundesliga' => 'bundesliga.json',
    'seriea'     => 'seriea.json',
    'ligue1'     => 'ligue1.json',
];
$soccer_standings_results = [];
foreach ($soccerStandingsMap as $lg => $file) {
    $ss_src = find_file($dir, $lg . '_standings_');
    if ($ss_src) { $used[basename($ss_src)] = true; }
    if (!$ss_src) {
        $soccer_standings_results[$lg] = ['ok' => true, 'skipped' => true, 'reason' => 'no file'];
        continue;
    }
    try {
        $ss_dst = '/var/www/sportsnews/assets/data/' . $file;
        list($ss_raw, $ss_new) = read_json($ss_src);
        $ss_in = $ss_new['standings'] ?? null;
        if (!is_array($ss_in) || count($ss_in) < 10) {
            $soccer_standings_results[$lg] = ['ok' => true, 'skipped' => true, 'file' => basename($ss_src), 'reason' => 'standings < 10'];
            continue;
        }
        if (!is_file($ss_dst)) {
            throw new RuntimeException($file . ' missing');
        }
        $ss_cur = json_decode(file_get_contents($ss_dst), true);
        if (!is_array($ss_cur)) {
            throw new RuntimeException($file . ' unreadable');
        }
        $ss_rows = [];
        foreach ($ss_in as $ss_s) {
            if (!is_array($ss_s)) continue;
            $ss_rows[] = [
                'rank'           => (int)($ss_s['rank'] ?? 0),
                'team'           => $ss_s['team_ko'] ?? $ss_s['team'] ?? '',
                'team_url'       => $ss_s['team_url'] ?? '',
                'qualification'  => $ss_s['qualification'] ?? '',
                'played'         => (int)($ss_s['played'] ?? 0),
                'win'            => (int)($ss_s['win'] ?? 0),
                'draw'           => (int)($ss_s['draw'] ?? 0),
                'loss'           => (int)($ss_s['loss'] ?? 0),
                'goals_for'      => (int)($ss_s['goals_for'] ?? 0),
                'goals_against'  => (int)($ss_s['goals_against'] ?? 0),
                'goal_diff'      => (int)($ss_s['goal_diff'] ?? 0),
                'points'         => (int)($ss_s['points'] ?? 0),
                'form'           => $ss_s['form_recent'] ?? $ss_s['form'] ?? [],
            ];
        }
        $ss_cur['standings'] = $ss_rows;
        if (!empty($ss_new['generated_at'])) {
            $ss_cur['collected_at'] = $ss_new['generated_at'];
        }
        $ss_out = json_encode($ss_cur, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES | JSON_PRETTY_PRINT);
        if (!$ss_out) throw new RuntimeException('json_encode fail');
        copy($ss_dst, $ss_dst . '.prev');
        file_put_contents($ss_dst, $ss_out);
        $soccer_standings_results[$lg] = [
            'ok'      => true,
            'file'    => basename($ss_src),
            'standings' => count($ss_rows),
        ];
    } catch (Exception $e) {
        $soccer_standings_results[$lg] = ['ok' => false, 'file' => basename($ss_src), 'error' => $e->getMessage()];
    }
}
```
